package reutilizacion;

public class Print {

    public static String imprimirTextoYRetornar(String msg) {
        System.out.print(msg);
        return msg;
    }

    public static void imprimirSaltoLinea() {
        System.out.println();
    }

    public static void imprimirSinSalto(String msg) {
        System.out.print(msg);
    }

    public static void imprimirConSalto(String msg) {
        System.out.println(msg);
    }

    public static void imprimirObjetoConSalto(Object objeto) {
        System.out.println(objeto);
    }

    public static void imprimirObjetoSinSalto(Object objeto) {
        System.out.print(objeto);
    }
}

package reutilizacion;

import java.util.Scanner;

public class Sc {

    public static Scanner sc = new Scanner(System.in);

    public static int nextInt(String mensaje) {
        Print.imprimirSinSalto(mensaje);
        int numero = sc.nextInt();
        sc.nextLine();
        return numero;
    }

    public static double nextDouble(String mensaje) {
        Print.imprimirSinSalto(mensaje);
        double numero = sc.nextDouble();
        sc.nextLine();
        return numero;
    }

    public static boolean nextBoolean(String mensaje) {
        Print.imprimirSinSalto(mensaje);
        boolean respuesta = sc.nextBoolean();
        sc.nextLine();
        return respuesta;
    }

    public static String nextLine(String mensaje) {
        Print.imprimirSinSalto(mensaje);
        String texto = sc.nextLine();
        return texto;
    }

    public static String nextScan() {
        String texto = sc.nextLine();
        return texto;
    }
}

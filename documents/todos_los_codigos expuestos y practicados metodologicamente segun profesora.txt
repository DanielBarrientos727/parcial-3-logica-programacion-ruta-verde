package ciclos;

public class AdivinaElNumero {

    public static void main (String[] args){

        int secreto = 7;

        jugar(secreto);

    }

    //el usuario puede tener max 5 intentos (1-10) es bajito, esta alto // 7 ---- 1 es bajito, 10 --- muy alto
    // termina: acierte o se

    public static void jugar (int secreto){

        boolean acerto = false;
        int intentos = 0;

        while (!acerto && intentos <5){

            int numero = Entrada.leerEntero("Intento " + (intentos + 1) + " - adivina el número (1-10): ");
            intentos ++;

            if (numero == secreto){
                acerto = true;
            } else if (numero < secreto) {
                System.out.println("Muy bajo");
            }else {
                System.out.println("Muy alto");
            }
        }

        if (acerto){
            System.out.println("¡Ganasteee en " + intentos + " intentos!!!");
        }else {
            System.out.println("Perdiste, el número secreto era " + secreto);
        }


    }



}

public class AnalizadorDeFrases {

    public static void main(String[] args) {

        String frase = Entrada.leerLinea("Escribe una frase sin espacio al final: ");

        System.out.println("Letras (con espacios): " + frase.length()); // I like = 6
        System.out.println("Palabras: " + contarPalabras(frase));
        System.out.println("Primera palabra: " + primeraPalabra(frase));

    }
    public static int contarPalabras(String frase){
        int espacios = 0;

        for ( int i = 0; i < frase.length(); i++){
            if (frase.charAt(i) == ' '){ // pregunta si es un espacio
                espacios++;
            }
        }
    return espacios + 1;

    }
    public static String primeraPalabra( String frase){
        int posicion = frase.indexOf(' '); // guarda en que casillero está el primer espacio o -1

        if(posicion == -1){
            return frase; //.  COMERGUSTAR
        } //.  me gusta comer. // me

        return frase.substring(0, posicion);
    }
}

package arreglos;

public class Boletin {

    public static void main(String[] args) {

        int estudiantes = leerPostivo("¿Cuántos estudiantes? ");
        int notasPorEstudiante = leerPostivo("¿Cuántas notas por estudiante? ");

        double [][] notas = new double[estudiantes][notasPorEstudiante];

        llenar(notas);
        promedioPorEstudiantes(notas);
        promedioPorNota(notas);
    }
    public static int leerPostivo(String mensaje){
        int numero = Entrada.leerEntero(mensaje);

        while ( numero <= 0){
            System.out.println("Debe ser mayor que 0. ");
            numero = Entrada.leerEntero(mensaje);
        }
        return numero;
    }

    public static double leerNotaValida(String mensaje) {
        double nota = Entrada.leerDecimal(mensaje);

        while (nota < 0 || nota > 5) {
            System.out.println("Nota inválida. Debe estar entre 0 y 5. ");
            nota = Entrada.leerDecimal(mensaje);
        }
        return nota;
    }
    public static void llenar(double[][] notas){
        for ( int fila = 0; fila < notas.length; fila++){ // 2= 0, 3, 4
            for (int col = 0; col < notas[fila].length; col++){
                notas[fila][col] = leerNotaValida("Nota" + (col + 1)  + "del estudiante " + (fila + 1) + ": ");
            }
        }
    }

    public static void promedioPorEstudiantes(double[][] notas) {
        for (int fila = 0; fila < notas.length; fila++) {
            double suma = 0;
            for (int col = 0; col < notas[fila].length; col++) {
                suma += notas[fila][col];
            }
            System.out.println("Promedio del estudiante " + (fila + 1) + ": " + suma / notas[fila].length);

        }
    }



    public static void promedioPorNota(double[][] notas){
        for ( int col = 0; col < notas[0].length; col++){
            double suma = 0;

            for (int fila = 0; fila < notas.length; fila++){
                suma += notas[fila][col];
            }
            System.out.println("Promedio de la nota " + (col + 1) + ": " + suma/ notas.length);
        }

    }
}

package arreglos;

public class BuscarNumero {

    public static void main (String[] args){

        int [] numeros = {12, 7, 25, 3, 18};
        //                 0. 1.  2. 3.  4

        int buscado = Entrada.leerEntero(" ¿Qué número quieres buscar ? "); // 25

        int posicion = buscar (numeros, buscado);

        if (posicion == -1){
            System.out.println("El " + buscado + " no está en el arreglo. ");
        } else {
            System.out.println("El " + buscado + " está en la posición " + posicion + ".");
        }
    }

    public static  int buscar ( int[] numeros, int buscado){
        for (int i = 0; i < numeros.length; i++){
            if(numeros[i] == buscado){
                return i;
            }
        }

        return -1;
    }


}

        double costo;

        if (esInternacional){
            //Internacional
            if(peso<= 2){
                costo=80000;
            }else{
                costo = 80000 + (peso - 2) *25000;
            }
        } else {
            // Envío Nacional
            if(peso<=2){
                costo = 15000;
            }else{
                costo= 15000 + (peso - 2 ) *5000;
            }
        }

        return costo;
    }

        if (n <2){
            return false;
        }

        mostrarResumen(notas);
    }

    public static void leerNotas(double[] notas){
        for (int i = 0; i < notas.length; i++){

            notas[i] = Entrada.leerDecimal("Nota del estudiante " + (i + 1) + ": ");
            //                                       Nota del estudiante 1:
            //
        }

    }

    public static void mostrarResumen (double[] notas ){

        double suma = 0;
        int aprobados = 0;


        for (int i = 0; i < notas.length; i++){
            suma += notas[i];

            if (notas[i] >= 3.0){

                aprobados ++;
            }
        }

        double promedio = suma / notas.length;

        System.out.println("Promedio del grupo: " + promedio);
        System.out.println("Aprobados: " +aprobados + " de " + notas.length);


    }

}

            double porcentajeDescuento = ingresarReal(teclado, "¿Qué porcentaje de descuento tienes (ej. 10)? ");
            double costoDomicilio = ingresarReal(teclado, "¿Cuánto cuesta el domicilio? ");

            double descuento = calcularDescuento(valorPedido, porcentajeDescuento);
            double subtotal = calcularSubtotal(valorPedido, descuento);
            double total = calcularTotal(subtotal, costoDomicilio);

            String mensaje = generarMensaje(valorPedido, descuento, subtotal, costoDomicilio, total);
            mostrarMensaje(mensaje);
        }

        public static double ingresarReal(Scanner teclado, String pregunta) {
            System.out.print(pregunta);
            return teclado.nextDouble();
        }

        public static double calcularDescuento(double valorPedido, double porcentaje) {
            return valorPedido * (porcentaje / 100);
        }

        public static double calcularSubtotal(double valorPedido, double descuento) {
            return valorPedido - descuento;
        }

        public static double calcularTotal(double subtotal, double costoDomicilio) {
            return subtotal + costoDomicilio;
        }

        public static String generarMensaje(double valorPedido, double descuento, double subtotal,
                                            double costoDomicilio, double total) {
            return "Valor del pedido: $" + valorPedido +
                    ", descuento: $" + descuento +
                    ", subtotal: $" + subtotal +
                    ", domicilio: $" + costoDomicilio +
                    ", total a pagar: $" + total;
        }

        public static void mostrarMensaje(String mensaje) {
            System.out.println(mensaje);
        }
    }

public class Rombo {

    public static void main(String[] args) {

        int n = Entrada.leerEntero("¿Cuántas filas tiene la mitad de arriba)");

        int [] estrellas = new int [2 * n - 1];

        for ( int i = 0; i < n ; i ++){
            estrellas[i] = 2 * i + 1;
            estrellas[estrellas.length -1 -i] = 2 * i + 1;
        }

        dibujar(estrellas, 2 * n - 1);
    }
    public static void dibujar ( int [] estrellas, int ancho){
        for (int fila = 0; fila < estrellas.length; fila++){
            int espacios = (ancho - estrellas[fila] / 2);

            for (int j = 0; j < espacios; j++){
                System.out.print(" ");
            }

            for ( int j = 0; j < estrellas[fila]; j++){
                System.out.print("*");
            }
            System.out.println();
        }
    }
}

        }
    }

    public static void totalPorSemana(int[][] ventas) {
        for (int semana = 0; semana < ventas[0].length; semana++) {
            int total = 0;

            for (int sede = 0; sede < ventas.length; sede++) {
                total += ventas[sede][semana];
            }
            System.out.println("Semana " + (semana + 1) + ": " + total);
        }
    }
